cmake_minimum_required(VERSION 3.20)
project(streetflow VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/web/index.html" STREETFLOW_HTML)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/examples/CITY.IN" STREETFLOW_CITY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/examples/CITY.LAYOUT" STREETFLOW_LAYOUT)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS web/index.html examples/CITY.IN examples/CITY.LAYOUT)
configure_file(src/assets.hpp.in generated/assets.hpp @ONLY)
add_library(streetflow_core src/graph.cpp src/layout.cpp src/routing.cpp src/simulation.cpp)
target_include_directories(streetflow_core PUBLIC src)
target_compile_options(streetflow_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(streetflow src/main.cpp src/http_server.cpp)
target_include_directories(streetflow PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(streetflow PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(streetflow PRIVATE streetflow_core Threads::Threads)
include(GNUInstallDirs)
install(TARGETS streetflow RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES examples/CITY.IN examples/CITY.LAYOUT DESTINATION ${CMAKE_INSTALL_DATADIR}/streetflow/examples)
include(CTest)
if(BUILD_TESTING)
  add_executable(streetflow_tests tests/simulation_tests.cpp)
  target_link_libraries(streetflow_tests PRIVATE streetflow_core)
  add_test(NAME simulation COMMAND streetflow_tests)
  add_executable(streetflow_layout_tests tests/layout_tests.cpp)
  target_link_libraries(streetflow_layout_tests PRIVATE streetflow_core)
  add_test(NAME layout COMMAND streetflow_layout_tests)
  find_program(NODE_EXECUTABLE node)
  if(NODE_EXECUTABLE)
    add_test(NAME road_geometry COMMAND ${NODE_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/geometry_tests.js ${CMAKE_CURRENT_SOURCE_DIR}/web/index.html)
  endif()
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME city_generator COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/generator_tests.py)
  endif()
endif()
